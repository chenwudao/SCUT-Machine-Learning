# Compute error rate, alpha and w
def compute_error(y, y_pred, w_i):
    '''
    Calculate the error rate of a weak classifier m. Arguments:
    y: actual target value
    y_pred: predicted value by weak classifier
    w_i: individual weights for each observation
    
    Note that all arrays should be the same length
    tips: can try to use np.not_equal(y, y_pred)
    '''
    ######### start ##########
    # Weighted error: sum of weights where prediction != true, divided by sum of weights
    mismatches = np.not_equal(y, y_pred).astype(int)
    numerator = np.sum(w_i * mismatches)
    denominator = np.sum(w_i)
    error = numerator / denominator if denominator != 0 else 0.0
    return error
    #########  end  ##########

def compute_alpha(error):
    '''
    Calculate the weight of a weak classifier m in the majority vote of the final classifier. This is called
    alpha in chapter 10.1 of The Elements of Statistical Learning. Arguments:
    error: error rate from weak classifier m
    tips: can try to use np.log()
    '''
    ######### start ##########
    # Clip error to avoid division by zero or log of zero
    eps = 1e-10
    error = np.minimum(np.maximum(error, eps), 1 - eps)
    alpha = np.log((1 - error) / error)
    return alpha
    #########  end  ##########

def update_weights(w_i, alpha, y, y_pred):
    ''' 
    Update individual weights w_i after a boosting iteration. Arguments:
    w_i: individual weights for each observation
    y: actual target value
    y_pred: predicted value by weak classifier  
    alpha: weight of weak classifier used to estimate y_pred
    tips: can try to use np.not_equal(y, y_pred) and np.exp()
    ''' 
    ######### start  ##########
    # Increase weights for misclassified observations and normalize
    mismatches = np.not_equal(y, y_pred).astype(int)
    w_new = w_i * np.exp(alpha * mismatches)
    # normalize to sum to 1 (or keep proportion)
    denom = np.sum(w_new)
    if denom == 0:
        return w_new
    return w_new / denom
    #########  end  ##########

# Define AdaBoost class
class AdaBoost:
    
    def __init__(self):
        self.alphas = []
        self.G_M = []
        self.M = None
        self.training_errors = []
        self.prediction_errors = []

    def fit(self, X, y, M = 100):
        '''
        Fit model. Arguments:
        X: independent variables - array-like matrix
        y: target variable - array-like vector
        M: number of boosting rounds. Default is 100 - integer
        '''
        
        # Clear before calling
        self.alphas = [] 
        self.training_errors = []
        self.M = M

        # Iterate over M weak classifiers
        for m in range(0, M):
            
            # Set weights for current boosting iteration
            ######### start ##########
            if m == 0:
                # At m = 0, weights are all the same and equal to 1 / N
                w_i = np.ones(len(y)) / len(y)
            else:
                # (d) Update w_i using previous classifier and alpha (from last iteration)
                y_prev = self.G_M[-1].predict(X)
                w_i = update_weights(w_i, self.alphas[-1], y, y_prev)
            #########  end  ##########
            
            # (a) Fit weak classifier and predict labels
            G_m = DecisionTreeClassifier(max_depth = 1)     # Stump: Two terminal-node classification tree
            G_m.fit(X, y, sample_weight = w_i)
            y_pred = G_m.predict(X)       #m-th weak classifier's predict 
            
            self.G_M.append(G_m) # Save to list of weak classifiers
            
            ######### start ##########
            # (b) Compute error
            error_m = compute_error(y, y_pred, w_i) #use the function you have written
            #########  end  ##########
            
            self.training_errors.append(error_m)

            # (c) Compute alpha
            alpha_m = compute_alpha(error_m)
            self.alphas.append(alpha_m)

        assert len(self.G_M) == len(self.alphas)
    def predict(self, X):
        '''
        Predict using fitted model. Arguments:
        X: independent variables - array-like
        '''

        # Initialise dataframe with weak predictions for each observation
        weak_preds = pd.DataFrame(index = range(len(X)), columns = range(self.M)) 

        # Predict class label for each weak classifier, weighted by alpha_m
        for m in range(self.M):
            
            ######### start ##########
            y_pred_m = self.G_M[m].predict(X)        # base on:(m-th weak classifier's predict:y_pred = G_m.predict(X))
            #########  end  ##########
            
            weak_preds.iloc[:,m] = y_pred_m   #each row corresponds to the ouput of m-th weak classifier

        # Calculate final predictions
        ######### start ##########
        # Weighted sum of weak predictions followed by sign to get final class {-1, 1}
        weak_arr = weak_preds.values.astype(float)
        alphas_arr = np.array(self.alphas)
        weighted = np.dot(weak_arr, alphas_arr)
        y_pred = np.sign(weighted)
        # Convert zeros to 1 (choose a class for exact zero sums)
        y_pred[y_pred == 0] = 1
        #########  end  ##########

        return y_pred
